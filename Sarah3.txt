function updateWeather() {
    let koordState = getState('0_userdata.0.Wetter.Koordinaten');
    let koordinatenWert = (koordState && koordState.val) ? koordState.val : '52.52,13.405';
    
    let teile = koordinatenWert.split(',').map(s => s.trim());
    let lat = teile[0] || '52.52';
    let lon = teile[1] || '13.405';

    // 1. Ortsnamen über OpenStreetMap (Nominatim) anhand der Koordinaten ermitteln
    let nominatimUrl = `https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}&zoom=10&accept-language=de`;

    axios.get(nominatimUrl, { headers: { 'User-Agent': 'ioBroker-SARAH-Script' } })
        .then(response => {
            if (response.data && response.data.address) {
                let addr = response.data.address;
                let gefundenerOrt = addr.city || addr.town || addr.village || addr.municipality || "Unbekannt";
                setState("0_userdata.0.Wetter.Ortsname", gefundenerOrt, true);
            }
        })
        .catch(err => {
            console.log("Konnte Ortsname nicht automatisch ermitteln: " + err.message);
        });

    // 2. Wetterdaten von Open-Meteo abrufen
    let openMeteoUrl = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,relative_humidity_2m,precipitation,wind_speed_10m,wind_direction_10m`;

    axios.get(openMeteoUrl)
        .then(response => {
            const current = response.data.current;
            setState("0_userdata.0.Wetter.Temperatur", current.temperature_2m, true);
            setState("0_userdata.0.Wetter.Feuchtigkeit", current.relative_humidity_2m, true);
            setState("0_userdata.0.Wetter.Niederschlag", current.precipitation, true);
            setState("0_userdata.0.Wetter.Windgeschwindigkeit", current.wind_speed_10m, true);
            setState("0_userdata.0.Wetter.Windrichtung", current.wind_direction_10m, true);
        })
        .catch(error => {
            console.error("Fehler beim Abrufen des Wetters: " + error);
        });
}

setInterval(updateWeather, 30 * 60 * 1000);
updateWeather();

function logTokenVerbrauch(response, bezeichnung) {
    if (response.data && response.data.usage) {
        let u = response.data.usage;
        let cacheHit = 0;
        let cacheMiss = u.prompt_tokens; // Fallback, falls Cache-Details fehlen

        // DeepSeek liefert die Cache-Werte auf oberster Ebene von `usage`.
        if (typeof u.prompt_cache_hit_tokens === 'number' && typeof u.prompt_cache_miss_tokens === 'number') {
            cacheHit = u.prompt_cache_hit_tokens;
            cacheMiss = u.prompt_cache_miss_tokens;
        } else if (u.prompt_tokens_details && typeof u.prompt_tokens_details.cached_tokens !== 'undefined') {
            // Fallback für OpenAI-kompatible Antwortformate.
            cacheHit = u.prompt_tokens_details.cached_tokens;
            cacheMiss = u.prompt_tokens - cacheHit;
        }

        console.log(`[DeepSeek Token (${bezeichnung})] Gesamt: ${u.total_tokens} | Prompt gesamt: ${u.prompt_tokens} (davon Cache: ${cacheHit} ⚡ / neu bezahlt: ${cacheMiss} 💸) | Antwort: ${u.completion_tokens}`);
    }
}

function getBenutzerKonfiguration(senderName) {
    for (let i = 0; i < USER_DATENPUNKTE.length; i++) {
        let state = getState(USER_DATENPUNKTE[i]);
        if (state && state.val) {
            try {
                let config = typeof state.val === 'string' ? JSON.parse(state.val) : state.val;
                if (config.username === senderName) {
                    if (!config.promptKey) config.promptKey = (i === 0) ? 'benutzer_eins' : 'benutzer_zwei'; 
                    return config;
                }
            } catch (e) {}
        }
    }
    return null;
}

async function fuehreMaxSuche(suchAnfrage, sender, anredeName) {
    let serperState = getState('0_userdata.0.SerperAPI');
    let SERPER_API_KEY = (serperState && serperState.val) ? serperState.val : '';
    console.log(">>> Max-Suche gestartet für: '" + suchAnfrage + "' | Key vorhanden: " + (SERPER_API_KEY ? "JA" : "NEIN"));
    if (!SERPER_API_KEY) {
        sendTo('telegram', { user: sender, text: "Max: Mein Websuche-Key fehlt in '0_userdata.0.SerperAPI'." });
        return;
    }

    try {
        let response = await axios({
            method: 'post',
            url: 'https://google.serper.dev/search',
            headers: { 'X-API-KEY': SERPER_API_KEY, 'Content-Type': 'application/json' },
            data: { q: suchAnfrage, gl: 'de', hl: 'de' },
            timeout: 10000
        });

        let data = response.data;
        let antwortText = "";

        if (data.answerBox && data.answerBox.snippet) {
            antwortText = data.answerBox.snippet;
        } else if (data.organic && data.organic.length > 0) {
            let treffer = data.organic[0];
            antwortText = treffer.snippet;
            if (treffer.link) antwortText += "\n\n(Quelle: " + treffer.link + ")";
        } else {
            antwortText = "Dazu habe ich im Internet leider nichts gefunden, " + anredeName + ".";
        }

        sendTo('telegram', { user: sender, text: "Max: " + antwortText });
    } catch (error) {
        console.error("Serper Fehler: " + error.message);
        sendTo('telegram', { user: sender, text: "Max: Uff, die Websuche ist fehlgeschlagen." });
    }
}

on({ id: 'telegram.0.communicate.request' }, async function (obj) {
    console.log(">>> TRIGGER AUSGELÖST! Wert: " + (obj.state ? obj.state.val : "kein state"));
    if (!obj.state || !obj.state.val) return;
    let rawVal = obj.state.val;
    if (typeof rawVal !== 'string') return;

    let match = rawVal.match(/^\[(.*?)\]([\s\S]*)$/);
    if (!match) return;

    let sender = match[1];
    let userText = match[2].trim();
    console.log(">>> 2. Absender: " + sender + " | Text: " + userText);

    if (!userText || userText.startsWith('/')) return;

    let currentUser = getBenutzerKonfiguration(sender);
    if (!currentUser) return;
    let anredeName = (currentUser.rufname && currentUser.rufname !== "rufname") ? currentUser.rufname : sender;

    if (/^max[\s,:]+/i.test(userText)) {
        console.log(">>> MAX-CHECK ERFOLGREICH GEGREIFT!");
        let direktSuchAnfrage = userText.replace(/^max[\s,:]+/i, '').trim();
        if (direktSuchAnfrage) {
            delete wartendeSuche[sender];
            await fuehreMaxSuche(direktSuchAnfrage, sender, anredeName);
            return;
        }
    }

    if (wartendeSuche[sender]) {
        let gemerkteFrage = wartendeSuche[sender];
        delete wartendeSuche[sender];

        let istJa = /^(ja|jo|jau|okay|ok|klar|mach|such|los|gerne|bitte)(?![a-zäöüß])/i.test(userText) &&
                    !/(nicht|nein|nee|nö|ne\b|lass|stopp|abbrechen)/i.test(userText);
        let istNein = /^(nein|nee|nö|ne|nicht|lass|stopp|abbrechen)(?![a-zäöüß])/i.test(userText);

        if (istJa) {
            await fuehreMaxSuche(gemerkteFrage, sender, anredeName);
            return;
        }
        if (istNein) {
            sendTo('telegram', { user: sender, text: "Sarah: Gut, dann lassen wir das eben." });
            return;
        }
        // Weder eindeutiges Ja noch eindeutiges Nein: als neue Nachricht weiterverarbeiten.
    }

    let deepseekState = getState('0_userdata.0.DeepseekAPI');
    let DEEPSEEK_API_KEY = (deepseekState && deepseekState.val) ? deepseekState.val : '';
    if (!DEEPSEEK_API_KEY) {
        sendTo('telegram', { user: currentUser.username, text: "Hoppla, mein API-Key fehlt!" });
        return;
    }

    let optimierterUserInhalt = userText;
    if (currentUser.rufname && currentUser.rufname !== "rufname") {
        optimierterUserInhalt = "[Nutzer-Rufname: " + currentUser.rufname + "]\n" + userText;
    }

    let jetztMs = Date.now();
    let memoryKey = currentUser.username;
    if (!chatVerläufe[memoryKey]) {
        chatVerläufe[memoryKey] = { messages: [], lastInteraction: jetztMs };
    } else if (jetztMs - chatVerläufe[memoryKey].lastInteraction > GEDÄCHTNIS_TIMEOUT_MS) {
        chatVerläufe[memoryKey].messages = [];
    }
    chatVerläufe[memoryKey].lastInteraction = jetztMs;
    let userVerlauf = chatVerläufe[memoryKey].messages;

    // Cache-optimierter Aufbau: Fester, universeller System-Prompt ganz vorne
    let messages = [
        { role: 'system', content: GLOBALE_REGELN_UND_FAKTEN + " " + SYSTEM_PROMPTS[currentUser.promptKey] }
    ];
    messages = messages.concat(userVerlauf);
    messages.push({ role: 'user', content: optimierterUserInhalt });

    try {
        let response = await axios({
            method: 'post',
            url: 'https://api.deepseek.com/chat/completions',
            headers: { 'Authorization': 'Bearer ' + DEEPSEEK_API_KEY, 'Content-Type': 'application/json' },
            data: { model: 'deepseek-chat', messages: messages, tools: toolsDefinition, tool_choice: "auto", stream: false },
            timeout: 15000
        });

        logTokenVerbrauch(response, "Anfrage");
        let responseMessage = response.data.choices[0].message;

        if (responseMessage.tool_calls && responseMessage.tool_calls.length > 0) {
            messages.push(responseMessage);
            for (let toolCall of responseMessage.tool_calls) {
                if (toolCall.function.name === "get_sensor_value") {
                    let requestedKey;
                    try {
                        let args = JSON.parse(toolCall.function.arguments);
                        requestedKey = args.sensor_name;
                    } catch (e) {
                        console.error(">>> Ungültige Tool-Argumente: " + toolCall.function.arguments);
                        messages.push({ role: "tool", tool_call_id: toolCall.id, name: "get_sensor_value", content: JSON.stringify({ fehler: "Ungültige Anfrage an das Sensor-Tool." }) });
                        continue;
                    }
                    let ergebnisDaten = {};

                    if (requestedKey === "aktuelle_zeit") {
                        let jetzt = new Date();
                        ergebnisDaten[requestedKey] = jetzt.toLocaleTimeString('de-DE', { hour: '2-digit', minute: '2-digit' }) + " Uhr, " + jetzt.toLocaleDateString('de-DE', { weekday: 'long', day: '2-digit', month: '2-digit', year: 'numeric' });
                    } else {
                        let eintrag = sensorMap[requestedKey];
                        if (eintrag) {
                            if (Array.isArray(eintrag)) {
                                for (let subSensor of eintrag) {
                                    let st = getState(subSensor.id);
                                    ergebnisDaten[subSensor.name] = (st && st.val !== null) ? st.val + " " + subSensor.unit : "unbekannt";
                                }
                            } else {
                                let st = getState(eintrag.id);
                                ergebnisDaten[requestedKey] = (st && st.val !== null) ? st.val + " " + eintrag.unit : "unbekannt";
                            }
                        } else {
                            ergebnisDaten["fehler"] = "Sensor nicht gefunden.";
                        }
                    }
                    messages.push({ role: "tool", tool_call_id: toolCall.id, name: "get_sensor_value", content: JSON.stringify(ergebnisDaten) });
                }
            }

            let secondResponse = await axios({
                method: 'post',
                url: 'https://api.deepseek.com/chat/completions',
                headers: { 'Authorization': 'Bearer ' + DEEPSEEK_API_KEY, 'Content-Type': 'application/json' },
                data: { model: 'deepseek-chat', messages: messages, stream: false },
                timeout: 15000
            });

            logTokenVerbrauch(secondResponse, "Nach Tool-Call");
            responseMessage = secondResponse.data.choices[0].message;
        }

        let aiAntwort = responseMessage.content || "";
        console.log(">>> DEEPSEEK ROH-ANTWORT: " + aiAntwort);

        let maxMatch = aiAntwort.match(/\[FRAGE_AN_MAX:\s*(.*?)\]/i);
        if (maxMatch) {
            let suchBegriff = maxMatch[1].trim();
            wartendeSuche[sender] = suchBegriff;
            let saubereFrageAntwort = aiAntwort.replace(/\[FRAGE_AN_MAX:.*?\]/g, '').trim();
            userVerlauf.push({ role: 'user', content: optimierterUserInhalt });
            userVerlauf.push({ role: 'assistant', content: saubereFrageAntwort });
            if (userVerlauf.length > MAX_VERLAUF_LÄNGE) {
                userVerlauf.splice(0, userVerlauf.length - MAX_VERLAUF_LÄNGE);
            }
            sendTo('telegram', { user: currentUser.username, text: saubereFrageAntwort });
            return;
        }

        // ==========================================
        // HARDWARE-STEUERUNG (MQTT)
        // ==========================================
        let nichtErreichteGeräte = [];

        try {
            async function schalteMqtt(name, daten, aktion) {
                let zielPayload = (aktion === "ON") ? daten.payloadOn : daten.payloadOff;
                console.log(">>> MQTT SET an (" + name + "): " + daten.id + " = " + zielPayload);
                try {
                    await setStateAsync(daten.id, { val: zielPayload, ack: false });
                } catch (e1) {
                    try {
                        await setStateAsync(daten.id, zielPayload);
                    } catch (e2) {
                        console.error(">>> FEHLER beim MQTT-Schalten von " + name + ": " + e2.message);
                        nichtErreichteGeräte.push(name);
                    }
                }
            }

            let cmdRegex = /\[CMD:\s*([a-zA-ZäöüÄÖÜß_\-]+)_(on|off)\s*\]/gi;
            let cmdMatch;
            let anzahlCmd = 0;

            while ((cmdMatch = cmdRegex.exec(aiAntwort)) !== null) {
                anzahlCmd++;
                let zielName = cmdMatch[1].toLowerCase();
                let aktion = cmdMatch[2].toUpperCase();
                console.log(">>> CMD ERKANNT! Ziel: " + zielName + " | Aktion: " + aktion);

                let eintrag = geräteMap[zielName];
                if (eintrag) {
                    if (Array.isArray(eintrag)) {
                        for (let devName of eintrag) {
                            if (geräteMap[devName]) {
                                await schalteMqtt(devName, geräteMap[devName], aktion);
                            }
                        }
                    } else {
                        await schalteMqtt(zielName, eintrag, aktion);
                    }
                } else {
                    console.log(">>> FEHLER: '" + zielName + "' existiert nicht in der geräteMap!");
                }
            }

            if (anzahlCmd === 0) {
                console.log(">>> KEIN [CMD]-Tag in der Antwort enthalten.");
            }
        } catch (hwError) {
            console.error(">>> Hardware-Fehler (MQTT): " + hwError.message);
        }

        if (nichtErreichteGeräte.length > 0) {
            aiAntwort += "\n\nHinweis: Konnte folgende(s) Gerät(e) per MQTT nicht ansteuern: " + nichtErreichteGeräte.map(g => "'" + g + "'").join(", ") + ".";
        }

        let saubereAntwort = aiAntwort.replace(/\[CMD:.*?\]/g, '').trim();

        userVerlauf.push({ role: 'user', content: optimierterUserInhalt });
        userVerlauf.push({ role: 'assistant', content: saubereAntwort });
        if (userVerlauf.length > MAX_VERLAUF_LÄNGE) {
            userVerlauf.splice(0, userVerlauf.length - MAX_VERLAUF_LÄNGE);
        }

        sendTo('telegram', { user: currentUser.username, text: saubereAntwort });

    } catch (error) {
        console.error(">>> API FEHLER: " + (error.response ? JSON.stringify(error.response.data) : error.message));
        sendTo('telegram', { user: currentUser.username, text: "Uff, da ist ein Fehler aufgetreten." });
    }
});
